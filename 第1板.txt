  </div>

  <script>
    // ========== 全局数据 ==========
    let price = 50000
    let balance = 100000
    let positions = JSON.parse(localStorage.getItem('positions')) || []
    let history = JSON.parse(localStorage.getItem('history')) || []
    const feeRate = 0.0005 // 手续费万分之五

    // 渲染页面
    function render() {
      document.getElementById('price').innerText = price.toFixed(2)
      document.getElementById('balance').innerText = balance.toFixed(2) + ' USDT'
      renderPositions()
      renderHistory()
      saveData()
    }

    // 保存数据
    function saveData() {
      localStorage.setItem('positions', JSON.stringify(positions))
      localStorage.setItem('history', JSON.stringify(history))
    }

    // 价格自动波动
    setInterval(() => {
      price += (Math.random() - 0.5) * 200
      if (price < 10000) price = 10000
      checkLiquidation()
      render()
    }, 1500)

    // ========== 开仓 ==========
    function openLong() {
      openPosition('long')
    }
    function openShort() {
      openPosition('short')
    }

    function openPosition(side) {
      const amount = Number(document.getElementById('amount').value)
      const leverage = Number(document.getElementById('leverage').value)
      
      if (amount <= 0 || amount > balance) {
        alert('余额不足')
        return
      }

      const fee = amount * feeRate
      const value = amount * leverage
      const margin = amount

      balance -= (amount + fee)

      positions.push({
        side,
        entry: price,
        amount,
        leverage,
        value,
        margin,
        time: new Date().toLocaleString()
      })

      history.push({
        type: '开' + side,
        price: price.toFixed(2),
        amount: amount.toFixed(2),
        time: new Date().toLocaleString()
      })

      render()
    }

    // ========== 平仓 ==========
    function closePosition(index) {
      const pos = positions[index]
      const pnl = pos.side === 'long' 
        ? (price - pos.entry) * pos.value / pos.entry
        : (pos.entry - price) * pos.value / pos.entry

      const fee = pos.amount * feeRate
      const total = pos.margin + pnl - fee
      balance += total

      history.push({
        type: '平' + pos.side,
        pnl: pnl.toFixed(2),
        time: new Date().toLocaleString()
      })

      positions.splice(index, 1)
      render()
    }

    // ========== 渲染持仓 ==========
    function renderPositions() {
      const el = document.getElementById('positions')
      el.innerHTML = ''

      positions.forEach((pos, i) => {
        const pnl = pos.side === 'long'
          ? (price - pos.entry) * pos.value / price
          : (pos.entry - price) * pos.value / price
          
        const pnlClass = pnl >= 0 ? 'pnl-profit' : 'pnl-loss'

        const div = document.createElement('div')
        div.className = 'position-card'
        div.innerHTML = `
          <div>
            <div class="pos-side ${pos.side}">${pos.side === 'long' ? '做多' : '做空'}</div>
            <div style="margin-top:6px;font-size:12px;color:#999">开仓: ${pos.entry.toFixed(2)}</div>
          </div>
          <div class="${pnlClass}">${pnl >= 0 ? '+' : ''}${pnl.toFixed(2)}</div>
          <button class="close-btn" onclick="closePosition(${i})">平仓</button>
        `
        el.appendChild(div)
      })
    }

    // ========== 历史订单 ==========
    function renderHistory() {
      const el = document.getElementById('history')
      el.innerHTML = ''
      history.slice(-8).forEach(h => {
        const div = document.createElement('div')
        div.className = 'position-card'
        div.style.fontSize = '13px'
        div.innerHTML = `
          <div>${h.type}</div>
          <div>${h.pnl ? h.pnl : h.price}</div>
        `
        el.appendChild(div)
      })
    }

    // ========== 强平 ==========
    function checkLiquidation() {
      for (let i = positions.length - 1; i >= 0; i--) {
        const pos = positions[i]
        const pnl = pos.side === 'long'
          ? (price - pos.entry) * pos.value / pos.entry
          : (pos.entry - price) * pos.value / pos.entry

        if (pnl <= -pos.margin * 0.8) {
          balance -= pos.margin * 0.8
          positions.splice(i, 1)
          history.push({ type: '⚠️强制平仓', time: new Date().toLocaleString() })
        }
      }
    }

    render()
  </script>
