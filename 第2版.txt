  <script>
    const symbols = ['BTCUSDT','ETHUSDT','SOLUSDT','XRPUSDT','DOGEUSDT'];
    let currentSymbol = 'BTCUSDT';
    let price = 0;
    let priceData = [];
    let balance = 100000;
    let positions = JSON.parse(localStorage.getItem('pos')) || [];
    let history = JSON.parse(localStorage.getItem('hist')) || [];
    let chart = null;

    function init() {
      renderSymbols();
      connectWS();
      fetchTicker();
      initChart();
      render();
    }

    function renderSymbols() {
      document.getElementById('symbols').innerHTML = symbols.map(s => 
        `<button class="symbol-btn ${s===currentSymbol?'active':''}" onclick="selectSymbol('${s}')">${s.replace('USDT','')}</button>`
      ).join('');
    }

    function selectSymbol(s) {
      currentSymbol = s;
      renderSymbols();
      fetchTicker();
    }

    function connectWS() {
      const ws = new WebSocket('wss://stream.binance.com:9443/ws/!ticker@arr');
      ws.onmessage = (e) => {
        JSON.parse(e.data).forEach(t => {
          if(t.s === currentSymbol) {
            price = parseFloat(t.c);
            updateDisplay(t);
            checkLiquidation();
            render();
          }
        });
      };
    }

    function fetchTicker() {
      fetch('https://api.binance.com/api/v3/ticker/24hr?symbol='+currentSymbol)
        .then(r => r.json())
        .then(d => {
          price = parseFloat(d.lastPrice);
          updateDisplay(d);
          fetchKlines();
        });
    }

    function updateDisplay(d) {
      document.getElementById('symbolName').textContent = currentSymbol.replace('USDT','/USDT');
      document.getElementById('price').textContent = d.lastPrice;
      const ch = parseFloat(d.priceChangePercent);
      document.getElementById('change').innerHTML = `<span style="color:${ch>=0?'#00D8A6':'#FF4757'}">${ch>=0?'+':''}${ch}%</span>`;
    }

    function fetchKlines() {
      fetch('https://api.binance.com/api/v3/klines?symbol='+currentSymbol+'&interval=1m&limit=60')
        .then(r => r.json())
        .then(d => {
          priceData = d.map(k => parseFloat(k[4]));
          updateChart();
        });
    }

    function initChart() {
      const ctx = document.getElementById('chart').getContext('2d');
      chart = new Chart(ctx, {
        type: 'line',
        data: {
          labels: Array(60).fill(''),
          datasets: [{
            label: 'Price',
            data: [],
            borderColor: '#00D8A6',
            backgroundColor: 'rgba(0,216,166,0.1)',
            fill: true,
            tension: 0.4,
            pointRadius: 0
          }]
        },
        options: {
          responsive: true,
          maintainAspectRatio: false,
          plugins: {legend: {display: false}},
          scales: {x: {display: false}, y: {display: false}}
        }
      });
    }

    function updateChart() {
      if(chart) {
        chart.data.datasets[0].data = priceData;
        chart.update();
      }
    }

    function showPage(p) {
      document.querySelectorAll('.page').forEach(el => el.classList.remove('active'));
      document.querySelectorAll('.tabbar-item').forEach(el => el.classList.remove('active'));
      document.getElementById('page-'+p).classList.add('active');
      event.target.classList.add('active');
    }

    function openPosition(side) {
      const amount = Number(document.getElementById('amount').value);
      const leverage = Number(document.getElementById('leverage').value);
      if(amount <= 0 || amount > balance) {
        alert('Insufficient balance');
        return;
      }
      const fee = amount * 0.0005;
      balance -= (amount + fee);
      
      positions.push({
        id: Date.now(),
        symbol: currentSymbol,
        side: side,
        entry: price,
        size: amount,
        leverage: leverage,
        margin: amount
      });
      
      history.unshift({type:'Open '+(side==='long'?'Long':'Short'), time:new Date().toLocaleString()});
      
      save();
      render();
    }

    function closePosition(i) {
      const pos = positions[i];
      const pnl = pos.side === 'long' 
        ? (price - pos.entry) * pos.size * pos.leverage / pos.entry
        : (pos.entry - price) * pos.size * pos.leverage / pos.entry;
      const fee = pos.size * 0.0005;
      balance += (pos.margin + pnl - fee);
      
      history.unshift({type:'Close '+(pos.side==='long'?'Long':'Short')+' PNL:'+(pnl>=0?'+':'')+pnl.toFixed(2), time:new Date().toLocaleString()});
      
      positions.splice(i, 1);
      save();
      render();
    }

    function checkLiquidation() {
      for(let i = positions.length - 1; i >= 0; i--) {
        const pos = positions[i];
        const pnl = pos.side === 'long' 
          ? (price - pos.entry) * pos.size * pos.leverage / pos.entry
          : (pos.entry - price) * pos.size * pos.leverage / pos.entry;
        if(pnl <= -pos.margin * 0.8) {
          history.unshift({type:'LIQUIDATED!', time:new Date().toLocaleString()});
          positions.splice(i, 1);
        }
      }
      save();
    }

    function save() {
      localStorage.setItem('pos', JSON.stringify(positions));
      localStorage.setItem('hist', JSON.stringify(history));
    }

    function render() {
      let unrealized = 0;
      positions.forEach(pos => {
        const pnl = pos.side === 'long' 
          ? (price - pos.entry) * pos.size * pos.leverage / pos.entry
          : (pos.entry - price) * pos.size * pos.leverage / pos.entry;
        unrealized += pnl;
      });
      
      document.getElementById('totalBalance').textContent = (balance + unrealized).toLocaleString('en-US', {minimumFractionDigits:2});
      document.getElementById('balance').textContent = balance.toFixed(0);
      
      const posEl = document.getElementById('positions');
      if(positions.length === 0) {
        posEl.innerHTML = '<div class="empty-state">No positions</div>';
      } else {
        posEl.innerHTML = positions.map((pos, i) => {
          const pnl = pos.side === 'long' 
            ? (price - pos.entry) * pos.size * pos.leverage / pos.entry
            : (pos.entry - price) * pos.size * pos.leverage / pos.entry;
          return `
            <div class="position-card">
              <div style="display:flex;justify-content:space-between;align-items:center;margin-bottom:10px">
                <span style="background:${pos.side==='long'?'#00D8A6':'#FF4757'};padding:4px 8px;border-radius:6px;font-size:12px;font-weight:bold;color:${pos.side==='long'?'#000':'#fff'}">
                  ${pos.side==='long'?'LONG':'SHORT'} ${pos.leverage}x
                </span>
                <span style="color:${pnl>=0?'#00D8A6':'#FF4757'};font-weight:bold;font-size:18px">
                  ${pnl>=0?'+':''}${pnl.toFixed(2)}
                </span>
              </div>
              <div style="display:grid;grid-template-columns:1fr 1fr;gap:8px;font-size:12px;color:#999">
                <div>Entry: ${pos.entry.toFixed(2)}</div>
                <div>Size: ${pos.size} USDT</div>
              </div>
              <button style="margin-top:10px;width:100%;padding:10px;background:#252836;border:none;border-radius:8px;color:#fff;cursor:pointer" onclick="closePosition(${i})">
                Close Position
              </button>
            </div>
          `;
        }).join('');
      }
      
      const histEl = document.getElementById('history');
      if(history.length === 0) {
        histEl.innerHTML = '<div class="empty-state">No history</div>';
      } else {
        histEl.innerHTML = history.slice(0, 20).map(h => `
          <div class="position-card">
            <div>${h.type}</div>
            <div style="font-size:12px;color:#999;margin-top:4px">${h.time}</div>
          </div>
        `).join('');
      }
    }

    init();
  </script>
